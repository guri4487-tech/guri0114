"""간단한 스터디 출석부 프로그램 (외부 라이브러리 없이 표준 라이브러리만 사용)

기능
  1) 출석 추가 : 날짜와 이름을 입력하면 기록이 쌓임
  2) 목록 보기 : 지금까지 쌓인 출석을 날짜순으로 보여줌
  3) 종료

출석 기록은 같은 폴더의 attendance.csv 파일에 저장되어,
프로그램을 다시 켜도 기록이 유지됩니다.
"""

import csv
import os
from datetime import datetime

# 기록을 저장할 파일 이름 (이 스크립트와 같은 폴더에 생성됨)
DATA_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)), "attendance.csv")


def load_records():
    """CSV 파일에서 출석 기록을 읽어 리스트로 돌려준다."""
    records = []
    if not os.path.exists(DATA_FILE):
        return records
    with open(DATA_FILE, "r", encoding="utf-8", newline="") as f:
        reader = csv.DictReader(f)
        for row in reader:
            records.append(row)
    return records


def save_record(date, name):
    """한 건의 출석 기록을 CSV 파일 끝에 덧붙인다."""
    file_exists = os.path.exists(DATA_FILE)
    with open(DATA_FILE, "a", encoding="utf-8", newline="") as f:
        writer = csv.writer(f)
        if not file_exists:            # 파일이 처음 만들어질 때만 헤더를 쓴다
            writer.writerow(["date", "name"])
        writer.writerow([date, name])


def normalize_date(text):
    """입력한 날짜 문자열을 YYYY-MM-DD 형식으로 검증·정규화한다.
    형식이 올바르면 정규화된 문자열을, 아니면 None을 돌려준다.
    빈 값이면 오늘 날짜로 처리한다.
    """
    text = text.strip()
    if text == "":
        return datetime.now().strftime("%Y-%m-%d")
    try:
        d = datetime.strptime(text, "%Y-%m-%d")
        return d.strftime("%Y-%m-%d")
    except ValueError:
        return None


def add_attendance():
    """사용자에게 날짜와 이름을 입력받아 출석을 추가한다."""
    raw_date = input("날짜를 입력하세요 (YYYY-MM-DD, 빈칸이면 오늘): ")
    date = normalize_date(raw_date)
    if date is None:
        print("⚠️  날짜 형식이 올바르지 않습니다. 예) 2024-05-20\n")
        return

    name = input("이름을 입력하세요: ").strip()
    if name == "":
        print("⚠️  이름이 비어 있어 추가하지 않았습니다.\n")
        return

    save_record(date, name)
    print(f"✅ 출석 추가 완료: {date} / {name}\n")


def list_attendance():
    """지금까지 쌓인 출석 기록을 날짜순으로 보여준다."""
    records = load_records()
    if not records:
        print("아직 출석 기록이 없습니다.\n")
        return

    # 날짜 → 이름 순으로 정렬
    records.sort(key=lambda r: (r["date"], r["name"]))

    print("\n===== 출석 목록 =====")
    print(f"{'번호':<4} {'날짜':<12} {'이름'}")
    print("-" * 28)
    for i, r in enumerate(records, start=1):
        print(f"{i:<4} {r['date']:<12} {r['name']}")
    print(f"\n총 {len(records)}건\n")


def main():
    """메뉴를 반복해서 보여주는 메인 루프."""
    menu = (
        "==== 스터디 출석부 ====\n"
        "1) 출석 추가\n"
        "2) 목록 보기\n"
        "3) 종료\n"
        "선택: "
    )
    while True:
        choice = input(menu).strip()
        if choice == "1":
            add_attendance()
        elif choice == "2":
            list_attendance()
        elif choice == "3":
            print("프로그램을 종료합니다. 수고하셨어요! 👋")
            break
        else:
            print("⚠️  1, 2, 3 중에서 선택해 주세요.\n")


if __name__ == "__main__":
    main()
